# Artisan portable runtime layout contract.
#
# One machine-readable description of the Phase 9 layout-only portable
# package proof. Every key listed below is required, must appear exactly
# once, and is validated fail-closed by //tests/packaging:portable_layout_test:
# unknown keys, duplicate keys, missing keys, and drifted values fail the
# contract instead of being ignored.
#
# Scope boundary: this contract proves LAYOUT ONLY. It does not declare an
# installer, archive producer, updater, signing system, release channel,
# distribution service, process supervisor, or authenticated handoff. Those
# remain deferred decisions (docs/decisions/NATIVE_PRODUCT_SCOPE.md).

# Exact schema version. Only the literal value 1 is defined.
schema_version = 1

# What the currently reached proof covers. Only layout-only is defined;
# end-to-end or clean-host workflow coverage is not claimed.
proof_scope = layout-only

# Stable root directory of the portable package.
package_root = artisan-editor

# Sibling executable leaves sitting directly inside the package root.
# The editor locates Forge as its sibling by these declared names.
editor_sibling = artisan-editor.exe
forge_sibling = artisan-forge.exe

# Optional member directories. A layout-only package ships neither, but
# when a future package ships runtime resources or license documents,
# they must use exactly these directory names at the package root.
resources_dir = resources
licenses_dir = licenses

# Where mutable runtime state lives. The package root stays immutable at
# runtime; writable state lives outside it.
mutable_state = outside-package-root

# Production resolution policy. Production code must never fall back to
# source-tree paths or depend on a Bazel runfiles layout, so both flags
# must be the literal value false. Runfiles resolution exists only inside
# the Bazel test harness.
source_tree_fallback = false
runfiles_fallback = false
