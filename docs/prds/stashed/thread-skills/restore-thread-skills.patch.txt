+import { Effect } from "effect";
+
+import type {
+	EngineSkillQueryEnvelope,
+	EngineSkillQueryResultEnvelope,
+	EngineSkillReport,
+	ProtocolErrorDetail,
+} from "@artisan/protocol";
+import { EngineRegistry, type Engine } from "@artisan/engines";
+
+import { RuntimeMetadata } from "../../../runtime/metadata";
+
+const max_skills_per_report = 128;
+const max_engines_per_catalog = 16;
+const skills_timeout = "5 seconds";
+
+/** Renders a short, secret-free human message for a failed catalog read. */
+function describe_skill_failure(cause: unknown): string {
+	if (cause === null || typeof cause !== "object" || !("_tag" in cause)) {
+		return "Skill catalog read failed.";
+	}
+
+	const tag = String((cause as { _tag: unknown })._tag);
+
+	if (tag === "TimeoutException" || tag === "TimeoutError") {
+		return "Skill catalog read timed out.";
+	}
+
+	const message = (cause as { message?: unknown }).message;
+
+	return typeof message === "string" && message.length > 0
+		? message
+		: `Skill catalog read failed (${tag}).`;
+}
+
+/**
+ * Reads one engine's catalog, absorbing failure into the report shape.
+ *
+ * Uncached, unlike usage: this is a local directory walk rather than a provider
+ * round trip, and a skill added or a hook edited between turns should show up on
+ * the next read rather than after a freshness window expires.
+ */
+function ReportSkills(engine: Engine): Effect.Effect<EngineSkillReport> {
+	const skills = engine.Skills;
+
+	if (skills === undefined) {
+		return Effect.succeed({
+			display_name: engine.Descriptor.display_name,
+			engine_id: engine.Descriptor.id,
+			skills: [],
+		});
+	}
+
+	return skills.pipe(
+		Effect.timeout(skills_timeout),
+		Effect.match({
+			onFailure: (cause): EngineSkillReport => ({
+				display_name: engine.Descriptor.display_name,
+				engine_id: engine.Descriptor.id,
+				failure: describe_skill_failure(cause),
+				skills: [],
+			}),
+			onSuccess: (catalog): EngineSkillReport => ({
+				display_name: engine.Descriptor.display_name,
+				engine_id: engine.Descriptor.id,
+				skills: catalog.skills.slice(0, max_skills_per_report),
+			}),
+		}),
+	);
+}
+
+export const MakeEngineSkillQueryHandler = Effect.gen(function* () {
+	const registry = yield* EngineRegistry;
+	const metadata = yield* RuntimeMetadata;
+
+	const Envelope = <Kind extends EngineSkillQueryResultEnvelope["kind"], Payload>(
+		query: EngineSkillQueryEnvelope,
+		kind: Kind,
+		payload: Payload,
+	) =>
+		Effect.gen(function* () {
+			const message_id = yield* metadata.MakeId("message");
+			const sent_at = yield* metadata.Now;
+
+			return {
+				correlation_id: query.message_id,
+				kind,
+				message_id,
+				origin: "backend" as const,
+				payload,
+				protocol_version: 1 as const,
+				schema_version: 1 as const,
+				sent_at,
+			};
+		});
+
+	const handlers = {
+		"engine.skill.query": (query: EngineSkillQueryEnvelope) =>
+			Effect.gen(function* () {
+				const engines = (yield* registry.List).filter(
+					(engine) =>
+						engine.Skills !== undefined &&
+						(query.payload.engine_id === undefined ||
+							engine.Descriptor.id === query.payload.engine_id),
+				);
+				const reports = yield* Effect.all(engines.map(ReportSkills), {
+					concurrency: "unbounded",
+				});
+				const fetched_at = yield* metadata.Now;
+
+				return yield* Envelope(query, "engine.skill.query.result", {
+					engines: reports.slice(0, max_engines_per_catalog),
+					fetched_at,
+				});
+			}),
+	};
+
+	return (
+		query: EngineSkillQueryEnvelope,
+	): Effect.Effect<EngineSkillQueryResultEnvelope, ProtocolErrorDetail> => {
+		switch (query.kind) {
+			case "engine.skill.query":
+				return handlers["engine.skill.query"](query);
+		}
+	};
+});
